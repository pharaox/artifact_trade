destroy_artifact_aniconist_effect = {
	$ARTIFACT$ = {
		save_scope_as = artifact # STA Use a permanent scope for the event
	}

	$DESTROYER$ = {
		if = {
			limit = {
				OR = {
					$DESTROYER$.rite = { rite_has_parameter = destroying_artifacts_is_pious }
					$PIETY_BY_DEFAULT$ = yes
				}
			}
			give_destroyed_artifact_piety = yes
		}
		if = {
			limit = {
				has_personal_tenet_flag = tenet_aniconism_destruction_fulfillment
			}
			change_spiritual_fulfillment = minor_spiritual_fulfillment_value
		}
		if = {
			limit = {
				$GIVE_GOLD$ = yes
			}
			add_gold = {
				if = {
					limit = {
						scope:artifact = { rarity = illustrious }
					}
					add = {
						value = 25
					}
				}
				else_if = {
					limit = {
						scope:artifact = { rarity = famed }
					}
					add = {
						value = 15
					}
				}
				else_if = {
					limit = {
						scope:artifact = { rarity = masterwork }
					}
					add = {
						value = 8
					}
				}
				else = {
					add = {
						value = 4
					}
				}

				# Boost the value up to something meaningful
				multiply = {
					value = 10
				}

				# multiply by durability percentage
				multiply = {
					add = scope:artifact.artifact_durability
					divide = scope:artifact.artifact_max_durability	
					divide = 2
					add = 0.5
				}

				#Nerf it to bits for LAAMPs
				if = {
					limit = { 
						$DESTROYER$ = { has_government = landless_adventurer_government }
					}
					multiply = {
						value = 0.1
					}
					round = yes #Unop: Move the round here from the multiply block since it will make it multiply by 0
				}
				#Nerf it a bit less for Merchants
				else_if = {
					limit = { 
						$DESTROYER$ = { is_landless_merchant = yes }
					}
					multiply = {
						value = 0.5
					}
				}
				if = {
					limit = {
						exists = scope:artifact.var:ai_to_ai_destroy_in_sieges # Trash artifact
					}
					max = {
						value = $DESTROYER$.tiny_gold_value
						multiply = 0.5
					}
				}
				else_if = {
					limit = {
						scope:artifact = { is_unique = yes }
					}
					max = {
						value = $DESTROYER$.medium_gold_value
					}
				}
				else = {
					max = {
						value = $DESTROYER$.minor_gold_value
					}
				}

				round = yes
				min = 1
			}
		}

		# STA: Trigger event
		save_scope_as = destroyer
		scope:artifact = {
			trigger_event = {
				on_action = at_on_artifact_destroyed
			}
		}

		# Living creatures are still removed, but a sale should not describe that as destruction.
		if = {
			limit = {
				$GIVE_GOLD$ = yes
				is_landless_merchant = yes
				scope:artifact = { artifact_type = creature }
			}
			scope:artifact = { save_scope_as = sold_creature }
			custom_tooltip = {
				text = sell_creature_artifact_effect
				destroy_owned_artifact = scope:artifact
			}
		}
		else = {
			destroy_owned_artifact = scope:artifact
		}
	}
}
